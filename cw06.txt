CREATE OR REPLACE FUNCTION get_annual_salary (
  p_employee_id IN employees.employee_id%TYPE
) RETURN NUMBER
IS
  v_salary         employees.salary%TYPE;
  v_commission_pct employees.commission_pct%TYPE;
  v_annual_salary  NUMBER;
BEGIN
  SELECT salary, commission_pct
  INTO v_salary, v_commission_pct
  FROM employees
  WHERE employee_id = p_employee_id;

  v_annual_salary := (v_salary * 12) + (v_salary * NVL(v_commission_pct, 0));

  RETURN v_annual_salary;
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    RAISE_APPLICATION_ERROR(-20002, 'Pracownik o podanym ID: ' || p_employee_id || ' nie istnieje.');
  WHEN OTHERS THEN
    RAISE_APPLICATION_ERROR(-20099, 'Wystąpił nieoczekiwany błąd: ' || SQLERRM);
END get_annual_salary;
/

CREATE OR REPLACE FUNCTION format_phone_number (
  p_phone_number IN VARCHAR2
) RETURN VARCHAR2
IS
  v_area_code    VARCHAR2(20);
  v_rest_of_number VARCHAR2(50);
  v_dot_position NUMBER;
BEGIN
  IF p_phone_number IS NULL THEN
    RETURN NULL;
  END IF;

  v_dot_position := INSTR(p_phone_number, '.');

  IF v_dot_position > 0 THEN
    v_area_code := SUBSTR(p_phone_number, 1, v_dot_position - 1);
    v_rest_of_number := SUBSTR(p_phone_number, v_dot_position + 1);
    RETURN '(' || v_area_code || ') ' || v_rest_of_number;
  ELSE
    RETURN p_phone_number;
  END IF;
EXCEPTION
  WHEN OTHERS THEN
    RAISE_APPLICATION_ERROR(-20099, 'Wystąpił nieoczekiwany błąd podczas formatowania numeru: ' || SQLERRM);
END format_phone_number;
/

CREATE OR REPLACE FUNCTION format_string_case (
  p_input_string IN VARCHAR2
) RETURN VARCHAR2
IS
  v_length NUMBER;
  v_first_char VARCHAR2(1);
  v_last_char VARCHAR2(1);
  v_middle_part VARCHAR2(4000);
BEGIN
  IF p_input_string IS NULL THEN
    RETURN NULL;
  END IF;

  v_length := LENGTH(p_input_string);

  IF v_length = 0 THEN
    RETURN '';
  ELSIF v_length = 1 THEN
    RETURN UPPER(p_input_string);
  ELSE
    v_first_char := UPPER(SUBSTR(p_input_string, 1, 1));
    v_last_char := UPPER(SUBSTR(p_input_string, -1, 1));
    IF v_length > 2 THEN
      v_middle_part := LOWER(SUBSTR(p_input_string, 2, v_length - 2));
      RETURN v_first_char || v_middle_part || v_last_char;
    ELSE
      RETURN v_first_char || v_last_char;
    END IF;
  END IF;
EXCEPTION
  WHEN OTHERS THEN
    RAISE_APPLICATION_ERROR(-20099, 'Wystąpił nieoczekiwany błąd podczas formatowania ciągu: ' || SQLERRM);
END format_string_case;
/

CREATE OR REPLACE FUNCTION pesel_to_dob (
  p_pesel IN VARCHAR2
) RETURN VARCHAR2
IS
  v_year_str  VARCHAR2(2);
  v_month_str VARCHAR2(2);
  v_day_str   VARCHAR2(2);
  v_encoded_month NUMBER;
  v_actual_month  NUMBER;
  v_year          NUMBER;
  v_century_prefix VARCHAR2(2);
  v_dob           DATE;
BEGIN
  IF LENGTH(p_pesel) <> 11 OR NOT REGEXP_LIKE(p_pesel, '^[0-9]{11}$') THEN
    RAISE_APPLICATION_ERROR(-20003, 'Nieprawidłowy format PESEL. Oczekiwano 11 cyfr.');
  END IF;

  v_year_str  := SUBSTR(p_pesel, 1, 2);
  v_month_str := SUBSTR(p_pesel, 3, 2);
  v_day_str   := SUBSTR(p_pesel, 5, 2);
  v_encoded_month := TO_NUMBER(v_month_str);

  IF v_encoded_month >= 81 AND v_encoded_month <= 92 THEN
    v_century_prefix := '18';
    v_actual_month := v_encoded_month - 80;
  ELSIF v_encoded_month >= 1 AND v_encoded_month <= 12 THEN
    v_century_prefix := '19';
    v_actual_month := v_encoded_month;
  ELSIF v_encoded_month >= 21 AND v_encoded_month <= 32 THEN 
    v_century_prefix := '20';
    v_actual_month := v_encoded_month - 20;
  ELSE
    RAISE_APPLICATION_ERROR(-20004, 'Nieprawidłowy kod miesiąca w numerze PESEL: ' || v_month_str || '. Oczekiwano wartości 01-12, 21-32 lub 81-92.');
  END IF;

  v_year := TO_NUMBER(v_century_prefix || v_year_str);

  IF v_year > 2099 THEN
      RAISE_APPLICATION_ERROR(-20007, 'PESEL wskazuje na datę urodzenia po roku 2099.');
  END IF;

  BEGIN
    v_dob := TO_DATE(v_year || LPAD(v_actual_month, 2, '0') || v_day_str, 'YYYYMMDD');
  EXCEPTION
    WHEN OTHERS THEN
      RAISE_APPLICATION_ERROR(-20005, 'Nieprawidłowa data urodzenia wygenerowana z PESEL: ' || v_year || '-' || LPAD(v_actual_month, 2, '0') || '-' || v_day_str || '. Sprawdź poprawność dnia.');
  END;

  RETURN TO_CHAR(v_dob, 'YYYY-MM-DD');

EXCEPTION
  WHEN OTHERS THEN
    IF SQLCODE BETWEEN -20000 AND -20999 THEN
        RAISE;
    ELSE
        RAISE_APPLICATION_ERROR(-20099, 'Wystąpił nieoczekiwany błąd w funkcji pesel_to_dob: ' || SQLERRM);
    END IF;
END pesel_to_dob;
/

CREATE OR REPLACE PACKAGE types_pkg AS
  TYPE country_stats_rec IS RECORD (
    employee_count NUMBER,
    department_count NUMBER
  );
END types_pkg;
/

CREATE OR REPLACE FUNCTION get_country_stats (
  p_country_name IN countries.country_name%TYPE
) RETURN types_pkg.country_stats_rec
IS
  v_stats types_pkg.country_stats_rec;
  v_country_id countries.country_id%TYPE;
BEGIN
  BEGIN
    SELECT country_id
    INTO v_country_id
    FROM countries
    WHERE country_name = p_country_name;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20006, 'Kraj o nazwie "' || p_country_name || '" nie istnieje.');
  END;

  SELECT COUNT(DISTINCT e.employee_id)
  INTO v_stats.employee_count
  FROM employees e
  JOIN departments d ON e.department_id = d.department_id
  JOIN locations l ON d.location_id = l.location_id
  WHERE l.country_id = v_country_id;

  SELECT COUNT(DISTINCT d.department_id)
  INTO v_stats.department_count
  FROM departments d
  JOIN locations l ON d.location_id = l.location_id
  WHERE l.country_id = v_country_id;

  RETURN v_stats;
EXCEPTION
  WHEN OTHERS THEN
    IF SQLCODE BETWEEN -20000 AND -20999 THEN
        RAISE;
    ELSE
        RAISE_APPLICATION_ERROR(-20099, 'Wystąpił nieoczekiwany błąd w get_country_stats: ' || SQLERRM);
    END IF;
END get_country_stats;
/

  nazwa             VARCHAR2(30),
  data_zamkniecia   DATE,
  ostatni_manager   VARCHAR2(100)
);

CREATE OR REPLACE TRIGGER trg_archive_department
AFTER DELETE ON departments
FOR EACH ROW
DECLARE
  v_manager_name VARCHAR2(100);
BEGIN
  IF :OLD.manager_id IS NOT NULL THEN
    BEGIN
      SELECT first_name || ' ' || last_name
      INTO v_manager_name
      FROM employees
      WHERE employee_id = :OLD.manager_id;
    EXCEPTION
      WHEN NO_DATA_FOUND THEN
        v_manager_name := 'Manager nieznany (ID: ' || :OLD.manager_id || ')';
    END;
  ELSE
    v_manager_name := 'Brak managera';
  END IF;

  INSERT INTO archiwum_departamentow (id, nazwa, data_zamkniecia, ostatni_manager)
  VALUES (:OLD.department_id, :OLD.department_name, SYSDATE, v_manager_name);
  
  DBMS_OUTPUT.PUT_LINE('Zarchiwizowano departament ID: ' || :OLD.department_id);
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Błąd w trg_archive_department: ' || SQLERRM);
END trg_archive_department;
/

CREATE OR REPLACE PROCEDURE log_salary_violation (
    p_user          IN VARCHAR2,
    p_action        IN VARCHAR2,
    p_employee_id   IN NUMBER,
    p_old_salary    IN NUMBER,
    p_new_salary    IN NUMBER
) IS
    PRAGMA AUTONOMOUS_TRANSACTION;
BEGIN
    INSERT INTO zlodziej (
        uzytkownik,
        czas_zmiany,
        akcja,
        employee_id_ref,
        stare_wynagrodzenie,
        nowe_wynagrodzenie
    ) VALUES (
        p_user,
        SYSTIMESTAMP,
        p_action,
        p_employee_id,
        p_old_salary,
        p_new_salary
    );
    COMMIT;
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Błąd w procedurze log_salary_violation: ' || SQLERRM);
        ROLLBACK;
        RAISE;
END log_salary_violation;
/

CREATE OR REPLACE TRIGGER trg_check_salary_range
BEFORE INSERT OR UPDATE OF salary ON employees
FOR EACH ROW
BEGIN
  IF :NEW.salary NOT BETWEEN 2000 AND 26000 THEN
    log_salary_violation(
        p_user          => USER,
        p_action        => CASE WHEN INSERTING THEN 'INSERT' ELSE 'UPDATE' END,
        p_employee_id   => :NEW.employee_id,
        p_old_salary    => :OLD.salary,
        p_new_salary    => :NEW.salary
    );

    RAISE_APPLICATION_ERROR(-20010, 'Wynagrodzenie (' || :NEW.salary || ') musi być w przedziale 2000 - 26000. Próba zmiany została zarejestrowana.');
  END IF;
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Nieoczekiwany błąd w trg_check_salary_range: ' || SQLERRM);
        RAISE;
END trg_check_salary_range;
/

CREATE OR REPLACE TRIGGER trg_keep_old_salaries_jobs
BEFORE UPDATE OF min_salary, max_salary ON jobs
FOR EACH ROW
BEGIN
  IF UPDATING('min_salary') AND (:NEW.min_salary <> :OLD.min_salary OR (:NEW.min_salary IS NULL AND :OLD.min_salary IS NOT NULL) OR (:NEW.min_salary IS NOT NULL AND :OLD.min_salary IS NULL) ) THEN
    DBMS_OUTPUT.PUT_LINE('Próba zmiany min_salary dla job_id ' || :OLD.job_id || ' z ' || NVL(TO_CHAR(:OLD.min_salary), 'NULL') || ' na ' || NVL(TO_CHAR(:NEW.min_salary), 'NULL') || '. Przywracam starą wartość.');
    :NEW.min_salary := :OLD.min_salary;
  END IF;

  IF UPDATING('max_salary') AND (:NEW.max_salary <> :OLD.max_salary OR (:NEW.max_salary IS NULL AND :OLD.max_salary IS NOT NULL) OR (:NEW.max_salary IS NOT NULL AND :OLD.max_salary IS NULL) ) THEN
    DBMS_OUTPUT.PUT_LINE('Próba zmiany max_salary dla job_id ' || :OLD.job_id || ' z ' || NVL(TO_CHAR(:OLD.max_salary), 'NULL') || ' na ' || NVL(TO_CHAR(:NEW.max_salary), 'NULL') || '. Przywracam starą wartość.');
    :NEW.max_salary := :OLD.max_salary;
  END IF;
END trg_keep_old_salaries_jobs;
/

    WHEN OTHERS THEN IF SQLCODE BETWEEN -20000 AND -20999 THEN RAISE; ELSE RAISE_APPLICATION_ERROR(-20099, 'Błąd w get_country_stats: ' || SQLERRM); END IF;
  END get_country_stats;

  PROCEDURE dodaj_job (p_job_id IN jobs.job_id%TYPE, p_job_title IN jobs.job_title%TYPE) IS
  BEGIN
    INSERT INTO jobs (job_id, job_title) VALUES (p_job_id, p_job_title); COMMIT;
    DBMS_OUTPUT.PUT_LINE('Dodano job: ' || p_job_id);
  EXCEPTION WHEN DUP_VAL_ON_INDEX THEN DBMS_OUTPUT.PUT_LINE('Job ID ' || p_job_id || ' już istnieje.'); ROLLBACK;
            WHEN OTHERS THEN DBMS_OUTPUT.PUT_LINE('Błąd dodaj_job: ' || SQLERRM); ROLLBACK;
  END dodaj_job;

  PROCEDURE modyfikuj_job_title (p_job_id IN jobs.job_id%TYPE, p_nowy_title IN jobs.job_title%TYPE) IS
    e_no_job_updated EXCEPTION; PRAGMA EXCEPTION_INIT(e_no_job_updated, -20001);
  BEGIN
    UPDATE jobs SET job_title = p_nowy_title WHERE job_id = p_job_id;
    IF SQL%NOTFOUND THEN RAISE e_no_job_updated; END IF; COMMIT;
    DBMS_OUTPUT.PUT_LINE('Zmodyfikowano job: ' || p_job_id);
  EXCEPTION WHEN e_no_job_updated THEN DBMS_OUTPUT.PUT_LINE('Job ID ' || p_job_id || ' nie znaleziony.'); ROLLBACK;
            WHEN OTHERS THEN DBMS_OUTPUT.PUT_LINE('Błąd modyfikuj_job_title: ' || SQLERRM); ROLLBACK;
  END modyfikuj_job_title;

  PROCEDURE usun_job (p_job_id IN jobs.job_id%TYPE) IS
    e_no_job_deleted EXCEPTION; PRAGMA EXCEPTION_INIT(e_no_job_deleted, -20002);
  BEGIN
    DELETE FROM jobs WHERE job_id = p_job_id;
    IF SQL%NOTFOUND THEN RAISE e_no_job_deleted; END IF; COMMIT;
    DBMS_OUTPUT.PUT_LINE('Usunięto job: ' || p_job_id);
  EXCEPTION WHEN e_no_job_deleted THEN DBMS_OUTPUT.PUT_LINE('Job ID ' || p_job_id || ' nie znaleziony do usunięcia.'); ROLLBACK;
            WHEN OTHERS THEN DBMS_OUTPUT.PUT_LINE('Błąd usun_job: ' || SQLERRM); ROLLBACK;
  END usun_job;

  PROCEDURE pobierz_dane_pracownika (p_employee_id IN employees.employee_id%TYPE, p_salary OUT employees.salary%TYPE, p_last_name OUT employees.last_name%TYPE) IS
  BEGIN
    SELECT salary, last_name INTO p_salary, p_last_name FROM employees WHERE employee_id = p_employee_id;
  EXCEPTION WHEN NO_DATA_FOUND THEN DBMS_OUTPUT.PUT_LINE('Pracownik ID ' || p_employee_id || ' nie znaleziony.'); p_salary := NULL; p_last_name := NULL;
            WHEN OTHERS THEN DBMS_OUTPUT.PUT_LINE('Błąd pobierz_dane_pracownika: ' || SQLERRM); p_salary := NULL; p_last_name := NULL;
  END pobierz_dane_pracownika;

  PROCEDURE dodaj_pracownika (
    p_first_name    IN employees.first_name%TYPE, p_last_name     IN employees.last_name%TYPE,
    p_email         IN employees.email%TYPE,    p_salary        IN employees.salary%TYPE,
    p_job_id        IN employees.job_id%TYPE,   p_hire_date     IN employees.hire_date%TYPE,
    p_department_id IN employees.department_id%TYPE, p_phone_number  IN employees.phone_number%TYPE,
    p_commission_pct IN employees.commission_pct%TYPE, p_manager_id    IN employees.manager_id%TYPE
  ) IS
    e_salary_too_high EXCEPTION; PRAGMA EXCEPTION_INIT(e_salary_too_high, -20003);
    v_new_employee_id employees.employee_id%TYPE;
  BEGIN
    IF p_salary > 20000 THEN RAISE e_salary_too_high; END IF;
    SELECT employees_autoinc_seq.NEXTVAL INTO v_new_employee_id FROM dual;
    INSERT INTO employees (employee_id, first_name, last_name, email, phone_number, hire_date, job_id, salary, commission_pct, manager_id, department_id)
    VALUES (v_new_employee_id, p_first_name, p_last_name, p_email, p_phone_number, p_hire_date, p_job_id, p_salary, p_commission_pct, p_manager_id, p_department_id);
    COMMIT; DBMS_OUTPUT.PUT_LINE('Dodano pracownika: ' || p_first_name || ' ' || p_last_name || ' ID: ' || v_new_employee_id);
  EXCEPTION WHEN e_salary_too_high THEN DBMS_OUTPUT.PUT_LINE('Wynagrodzenie ' || p_salary || ' za wysokie.'); ROLLBACK;
            WHEN DUP_VAL_ON_INDEX THEN DBMS_OUTPUT.PUT_LINE('Email ' || p_email || ' już istnieje.'); ROLLBACK;
            WHEN OTHERS THEN DBMS_OUTPUT.PUT_LINE('Błąd dodaj_pracownika: ' || SQLERRM); ROLLBACK;
  END dodaj_pracownika;

CREATE OR REPLACE PACKAGE regions_crud_pkg AS

  TYPE region_row_type IS RECORD (
    region_id regions.region_id%TYPE,
    region_name regions.region_name%TYPE
  );
  TYPE region_table_type IS TABLE OF region_row_type INDEX BY BINARY_INTEGER;

  PROCEDURE add_region (
    p_region_id   IN regions.region_id%TYPE,
    p_region_name IN regions.region_name%TYPE
  );
  PROCEDURE add_region (
    p_region_name IN regions.region_name%TYPE,
    p_new_region_id OUT regions.region_id%TYPE
  );

  FUNCTION get_region_by_id (
    p_region_id IN regions.region_id%TYPE
  ) RETURN regions%ROWTYPE;

  FUNCTION get_region_by_name (
    p_region_name IN regions.region_name%TYPE
  ) RETURN regions%ROWTYPE;

  FUNCTION get_regions_by_name_pattern (
    p_name_pattern IN VARCHAR2
  ) RETURN region_table_type;

  FUNCTION get_all_regions
  RETURN region_table_type;

  FUNCTION get_all_regions_cursor
  RETURN SYS_REFCURSOR;

  PROCEDURE update_region_name (
    p_region_id   IN regions.region_id%TYPE,
    p_new_name    IN regions.region_name%TYPE
  );

  PROCEDURE delete_region (
    p_region_id IN regions.region_id%TYPE
  );

  e_region_not_found EXCEPTION;
  PRAGMA EXCEPTION_INIT(e_region_not_found, -20100);
  e_region_name_exists EXCEPTION;
  PRAGMA EXCEPTION_INIT(e_region_name_exists, -20101);

END regions_crud_pkg;

CREATE SEQUENCE regions_pkg_seq START WITH 100 NOCACHE;

CREATE OR REPLACE PACKAGE BODY regions_crud_pkg AS

  PROCEDURE add_region (p_region_id IN regions.region_id%TYPE, p_region_name IN regions.region_name%TYPE) IS
  BEGIN
    INSERT INTO regions (region_id, region_name) VALUES (p_region_id, p_region_name);
    COMMIT;
  EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
      RAISE_APPLICATION_ERROR(-20102, 'Region o ID ' || p_region_id || ' już istnieje.');
    WHEN OTHERS THEN
      RAISE_APPLICATION_ERROR(-20199, 'Błąd add_region(id,name): ' || SQLERRM);
  END add_region;

  PROCEDURE add_region (p_region_name IN regions.region_name%TYPE, p_new_region_id OUT regions.region_id%TYPE) IS
    v_temp_id regions.region_id%TYPE;
  BEGIN
    BEGIN
        SELECT region_id INTO v_temp_id FROM regions WHERE region_name = p_region_name;
        RAISE e_region_name_exists;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN NULL;
        WHEN e_region_name_exists THEN RAISE;
    END;

    SELECT regions_pkg_seq.NEXTVAL INTO p_new_region_id FROM dual;
    INSERT INTO regions (region_id, region_name) VALUES (p_new_region_id, p_region_name);
    COMMIT;
  EXCEPTION
    WHEN e_region_name_exists THEN
      RAISE_APPLICATION_ERROR(-20101, 'Region o nazwie "' || p_region_name || '" już istnieje.');
    WHEN OTHERS THEN
      RAISE_APPLICATION_ERROR(-20199, 'Błąd add_region(name): ' || SQLERRM);
  END add_region;

  FUNCTION get_region_by_id (p_region_id IN regions.region_id%TYPE) RETURN regions%ROWTYPE IS
    v_region_row regions%ROWTYPE;
  BEGIN
    SELECT * INTO v_region_row FROM regions WHERE region_id = p_region_id;
    RETURN v_region_row;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN RAISE e_region_not_found;
  END get_region_by_id;

  FUNCTION get_region_by_name (p_region_name IN regions.region_name%TYPE) RETURN regions%ROWTYPE IS
    v_region_row regions%ROWTYPE;
  BEGIN
    SELECT * INTO v_region_row FROM regions WHERE region_name = p_region_name;
    RETURN v_region_row;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN RAISE e_region_not_found;
    WHEN TOO_MANY_ROWS THEN
        RAISE_APPLICATION_ERROR(-20103, 'Znaleziono wiele regionów o nazwie: ' || p_region_name);
  END get_region_by_name;

  FUNCTION get_regions_by_name_pattern (p_name_pattern IN VARCHAR2) RETURN region_table_type IS
    v_regions region_table_type;
    i BINARY_INTEGER := 0;
  BEGIN
    FOR rec IN (SELECT region_id, region_name FROM regions WHERE region_name LIKE p_name_pattern ORDER BY region_name) LOOP
      i := i + 1;
      v_regions(i).region_id := rec.region_id;
      v_regions(i).region_name := rec.region_name;
    END LOOP;
    RETURN v_regions;
  END get_regions_by_name_pattern;


  FUNCTION get_all_regions RETURN region_table_type IS
    v_regions region_table_type;
    i BINARY_INTEGER := 0;
  BEGIN
    FOR rec IN (SELECT region_id, region_name FROM regions ORDER BY region_name) LOOP
      i := i + 1;
      v_regions(i).region_id := rec.region_id;
      v_regions(i).region_name := rec.region_name;
    END LOOP;
    RETURN v_regions;
  END get_all_regions;

  FUNCTION get_all_regions_cursor RETURN SYS_REFCURSOR IS
    v_cursor SYS_REFCURSOR;
  BEGIN
    OPEN v_cursor FOR SELECT region_id, region_name FROM regions ORDER BY region_id;
    RETURN v_cursor;
  END get_all_regions_cursor;


  PROCEDURE update_region_name (p_region_id IN regions.region_id%TYPE, p_new_name IN regions.region_name%TYPE) IS
    v_temp_id regions.region_id%TYPE;
  BEGIN
    BEGIN
        SELECT region_id INTO v_temp_id FROM regions WHERE region_name = p_new_name AND region_id <> p_region_id;
        RAISE e_region_name_exists;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN NULL; -- OK
        WHEN e_region_name_exists THEN RAISE;
    END;

    UPDATE regions SET region_name = p_new_name WHERE region_id = p_region_id;
    IF SQL%NOTFOUND THEN RAISE e_region_not_found; END IF;
    COMMIT;
  EXCEPTION
    WHEN e_region_not_found THEN
        RAISE_APPLICATION_ERROR(-20100, 'Region o ID ' || p_region_id || ' nie znaleziony do aktualizacji.');
    WHEN e_region_name_exists THEN
        RAISE_APPLICATION_ERROR(-20101, 'Region o nazwie "' || p_new_name || '" już istnieje.');
    WHEN OTHERS THEN
        RAISE_APPLICATION_ERROR(-20199, 'Błąd update_region_name: ' || SQLERRM);
  END update_region_name;

  PROCEDURE delete_region (p_region_id IN regions.region_id%TYPE) IS
  BEGIN
    DELETE FROM regions WHERE region_id = p_region_id;
    IF SQL%NOTFOUND THEN RAISE e_region_not_found; END IF;
    COMMIT;
  EXCEPTION
    WHEN e_region_not_found THEN
        RAISE_APPLICATION_ERROR(-20100, 'Region o ID ' || p_region_id || ' nie znaleziony do usunięcia.');
    WHEN OTHERS THEN 
        RAISE_APPLICATION_ERROR(-20199, 'Błąd delete_region: ' || SQLERRM || '. Sprawdź, czy nie ma powiązanych krajów.');
  END delete_region;

END regions_crud_pkg;
